% -----------------------------------------------------------------------------

\subsection{Evolution Terms}

\begin{description}[style=nextline,leftmargin=4.2cm,font=\normalfont\bfseries]
  \item[NEAT] NeuroEvolution of Augmenting Topologies, the evolutionary algorithm used to evolve
  both weights and neural network structure.

  \item[Genome] The encoded neural-network representation of an agent, including node and
  connection genes.

  \item[Species] A compatibility-based grouping of similar genomes used to preserve innovation
  during evolution.

  \item[Representative genome] A selected genome snapshot exported to describe one species or one
  agent at a reporting point.
\end{description}

\subsection{Runtime and Analysis Terms}

\begin{description}[style=nextline,leftmargin=4.2cm,font=\normalfont\bfseries]
  \item[Readback] Copying compact simulation state from GPU memory to host memory for UI or
  export.

  \item[Report interval] The tick cadence at which structured output artifacts are refreshed and
  written.

  \item[Speed multiplier] The UI-side visualization cadence that controls how many simulation
  ticks are advanced per visible frame.

  \item[Render snapshot] The compact host-visible population data used to draw the world in the
  UI.

  \item[Run queue] The UI-managed list of pending experiments that execute sequentially.

  \item[GPU-first architecture] A design in which the dominant simulation state and hot
  execution path remain on the GPU, while the CPU mainly orchestrates and exports compact
  results.
\end{description}
